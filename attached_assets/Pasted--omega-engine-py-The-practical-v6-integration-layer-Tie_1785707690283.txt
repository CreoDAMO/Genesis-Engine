@dataclass

    """Production configuration for the Omega Engine."""
    
    # Polymarket
    pm_api_key: str = field(default_factory=lambda: os.getenv("PM_API_KEY", ""))
    pm_api_secret: str = field(default_factory=lambda: os.getenv("PM_API_SECRET", ""))
    pm_passphrase: str = field(default_factory=lambda: os.getenv("PM_PASSPHRASE", ""))
    pm_rpc_url: str = "https://polygon-rpc.com"
    
    # Deribit (for perp hedge)
    deribit_key: str = field(default_factory=lambda: os.getenv("DERIBIT_KEY", ""))
    deribit_secret: str = field(default_factory=lambda: os.getenv("DERIBIT_SECRET", ""))
    
    # Trading parameters
    markets: List[str] = field(default_factory=lambda: [
        "0x...",  # Replace with actual PM market IDs
    ])
    max_position_usd: float = 5_000.0
    target_spread: float = 0.02
    quote_interval_sec: float = 3.0
    
    # Risk
    max_drawdown_pct: float = 0.15
    daily_loss_limit_usd: float = 500.0
    emergency_hedge_threshold: float = 0.8
    
    # GP / Strategy
    gp_population_size: int = 100
    gp_generations: int = 20
    gp_elite_pct: float = 0.05
    wasm_fuel_limit: int = 10_000
    
    # Audit
    audit_log_path: str = "logs/audit_omega.jsonl"
    
    # Mode
    paper_trade: bool = True  # START HERE. Set False only after weeks of paper.


# ------------------------------------------------------------------------------
# Omega Engine
# ------------------------------------------------------------------------------

class OmegaEngine:
    """
    The practical v6 trading engine.
    
    Architecture:
        ┌─────────────┐     ┌──────────────┐     ┌─────────────┐
        │  Deribit    │────▶│   Reality    │────▶│   Gravity   │
        │   WS Feed   │     │   Surface    │     │     LP      │
        └─────────────┘     └──────────────┘     └──────┬──────┘
        ┌─────────────┐            ▲                    │
        │ Polymarket  │────────────┘                    │
        │   WS Feed   │                                 │
        └─────────────┘                                 ▼
                                               ┌──────────────┐
                                               │  Polymarket  │
                                               │    CLOB      │
                                               └──────────────┘
        ┌─────────────┐     ┌──────────────┐
        │    GP       │────▶│    WASM      │
        │  Engine     │     │  Sandbox     │
        └─────────────┘     └──────────────┘
    """
    
    def __init__(self, config: Optional[OmegaConfig] = None):
        self.cfg = config or OmegaConfig()
        self._shutdown_event = asyncio.Event()
        
        # Subsystems (initialized in start())
        self.surface = None
        self.gravity = None
        self.wasm = None
        self.pm_client = None
        self.deribit_client = None
        self.gp_engine = None
        
        # State
        self.daily_pnl: float = 0.0
        self.peak_pnl: float = 0.0
        self.session_start: float = time.time()
        self.is_running: bool = False
        
        # Performance tracking
        self.latency_samples: deque = deque(maxlen=1000)
        self.quote_count: int = 0
        self.fill_count: int = 0
        
        logger.info("OmegaEngine initialized")
        logger.info(f"  Paper trade: {self.cfg.paper_trade}")
        logger.info(f"  Markets: {len(self.cfg.markets)}")
        logger.info(f"  Max position: ${self.cfg.max_position_usd:,.0f}")
    
    # ------------------------------------------------------------------
    # Lifecycle
    # ------------------------------------------------------------------
    
    async def start(self):
        """Initialize all subsystems and start the event loop."""
        logger.info("=" * 60)
        logger.info("OMEGA ENGINE STARTING")
        logger.info("=" * 60)
        
        # 1. Initialize WASM compiler (safe sandbox)
        from wasm_compiler import WASMStrategyCompiler
        self.wasm = WASMStrategyCompiler(fuel_limit=self.cfg.wasm_fuel_limit)
        logger.info(f"[WASM] Compiler ready (fuel={self.cfg.wasm_fuel_limit})")
        
        # 2. Initialize Reality Surface
        from reality_surface_claim_normalizer import RealitySurface
        self.surface = RealitySurface()
        logger.info("[Surface] Reality surface initialized")
        
        # 3. Initialize Polymarket client (your existing connector)
        # self.pm_client = PolymarketCLOB(...)  # Plug in your existing client
        logger.info("[PM] Client ready (placeholder)")
        
        # 4. Initialize Deribit client (your existing connector)
        # self.deribit_client = DeribitAPI(...)  # Plug in your existing client
        logger.info("[Deribit] Client ready (placeholder)")
        
        # 5. Initialize Gravity LP
        from gravity_lp_dominance import GravityMarketMaker
        self.gravity = GravityMarketMaker(
            polymarket_client=self.pm_client,
            reality_surface=self.surface,
            perp_hedger=self._hedge_on_deribit if not self.cfg.paper_trade else None,
            target_spread=self.cfg.target_spread,
            max_inventory=self.cfg.max_position_usd,
            emergency_hedge_threshold=self.cfg.emergency_hedge_threshold,
        )
        logger.info("[Gravity] LP engine initialized")
        
        # 6. Initialize GP engine (your existing genetic programming)
        # self.gp_engine = GeneticStrategyEngine(...)
        logger.info("[GP] Strategy engine ready (placeholder)")
        
        # 7. Setup signal handlers
        for sig in (signal.SIGINT, signal.SIGTERM):
            asyncio.get_event_loop().add_signal_handler(sig, self._signal_handler)
        
        self.is_running = True
        
        # Start main loops
        await asyncio.gather(
            self._price_ingestion_loop(),
            self._quoting_loop(),
            self._risk_monitor_loop(),
            self._gp_evolution_loop(),
            self._audit_loop(),
        )
    
    def _signal_handler(self):
        """Graceful shutdown on Ctrl+C."""
        logger.warning("Shutdown signal received")
        self._shutdown_event.set()
    
    async def stop(self):
        """Graceful shutdown."""
        logger.info("=" * 60)
        logger.info("OMEGA ENGINE SHUTTING DOWN")
        logger.info("=" * 60)
        self.is_running = False
        
        # Cancel all orders
        if self.gravity:
            for market_id in self.cfg.markets:
                await self.gravity._cancel_all(market_id)
        
        # Final audit
        self._write_audit({
            "event": "shutdown",
            "session_pnl": self.daily_pnl,
            "quotes": self.quote_count,
            "fills": self.fill_count,
            "duration_sec": time.time() - self.session_start
        })
        
        logger.info(f"Session PnL: ${self.daily_pnl:+.2f}")
        logger.info(f"Quotes placed: {self.quote_count}")
        logger.info(f"Fills: {self.fill_count}")
        logger.info("Goodbye.")
    
    # ------------------------------------------------------------------
    # Core Loops
    # ------------------------------------------------------------------
    
    async def _price_ingestion_loop(self):
        """
        Ingest prices from all venues every 1 second.
        This is the data backbone — everything else depends on fresh prices.
        """
        logger.info("[Loop] Price ingestion started")
        
        while not self._shutdown_event.is_set():
            try:
                t0 = time.time()
                
                # --- Polymarket ---
                for market_id in self.cfg.markets:
                    # In production: fetch from PM WebSocket or REST
                    pm_price = 0.62  # Replace with actual fetch
                    pm_liquidity = 150_000
                    
                    self.surface.ingest_polymarket(
                        event=market_id,
                        yes_price=pm_price,
                        liquidity_usd=pm_liquidity,
                        spread=0.02
                    )
                
                # --- Deribit ---
                # In production: fetch BTC perp + options
                # self.surface.ingest_deribit_option(...)
                # self.surface.ingest_funding_rate(...)
                
                # --- Check for divergences ---
                for market_id in self.cfg.markets:
                    div = self.surface.find_divergence(market_id, threshold=0.025)
                    if div:
                        logger.info(
                            f"[ARB] {market_id}: {div.direction} | "
                            f"edge={div.edge_bps}bps | conf={div.confidence:.2f}"
                        )
                
                latency = (time.time() - t0) * 1000
                self.latency_samples.append(latency)
                
                await asyncio.sleep(1.0)
                
            except Exception as e:
                logger.error(f"[PriceLoop] Error: {e}")
                await asyncio.sleep(5.0)
        
        logger.info("[Loop] Price ingestion stopped")
    
    async def _quoting_loop(self):
        """
        Refresh quotes on Polymarket every quote_interval_sec.
        This is where the engine makes money as a market maker.
        """
        logger.info("[Loop] Quoting started")
        
        while not self._shutdown_event.is_set():
            try:
                for market_id in self.cfg.markets:
                    if self.cfg.paper_trade:
                        # Paper mode: just log what we WOULD do
                        quotes = self.gravity.compute_quotes(market_id)
                        if quotes:
                            self.quote_count += 1
                            logger.debug(
                                f"[PAPER] {market_id} | "
                                f"BID {quotes['bid'].price:.4f}x{quotes['bid'].size:.0f} | "
                                f"ASK {quotes['ask'].price:.4f}x{quotes['ask'].size:.0f} | "
                                f"fair={quotes['fair']:.4f}"
                            )
                    else:
                        # Live mode: actually place orders
                        await self.gravity.refresh_quotes(market_id)
                        self.quote_count += 1
                
                await asyncio.sleep(self.cfg.quote_interval_sec)
                
            except Exception as e:
                logger.error(f"[QuoteLoop] Error: {e}")
                await asyncio.sleep(self.cfg.quote_interval_sec)
        
        logger.info("[Loop] Quoting stopped")
    
    async def _risk_monitor_loop(self):
        """
        Monitor risk metrics every 10 seconds.
        
        Kill switch triggers:
        - Daily loss > limit
        - Drawdown > max_drawdown_pct
        - Latency spike > 500ms
        """
        logger.info("[Loop] Risk monitor started")
        
        while not self._shutdown_event.is_set():
            try:
                # Check daily loss
                if self.daily_pnl < -self.cfg.daily_loss_limit_usd:
                    logger.critical(
                        f"[KILL SWITCH] Daily loss ${self.daily_pnl:+.2f} exceeds "
                        f"limit ${self.cfg.daily_loss_limit_usd}"
                    )
                    await self._emergency_stop()
                    return
                
                # Check drawdown
                self.peak_pnl = max(self.peak_pnl, self.daily_pnl)
                drawdown = self.peak_pnl - self.daily_pnl
                if self.peak_pnl > 0 and drawdown / self.peak_pnl > self.cfg.max_drawdown_pct:
                    logger.critical(
                        f"[KILL SWITCH] Drawdown {drawdown/self.peak_pnl:.1%} exceeds limit"
                    )
                    await self._emergency_stop()
                    return
                
                # Check latency
                if self.latency_samples:
                    p99_latency = np.percentile(self.latency_samples, 99)
                    if p99_latency > 500:
                        logger.warning(
                            f"[RISK] P99 latency spike: {p99_latency:.1f}ms"
                        )
                
                # Log status
                status = self.gravity.get_status() if self.gravity else {}
                logger.info(
                    f"[RISK] PnL=${self.daily_pnl:+.2f} | "
                    f"DD={drawdown:.2f} | "
                    f"Inv={status.get('total_net_delta', 0):.1f} | "
                    f"Quotes={self.quote_count} | "
                    f"Fills={self.fill_count}"
                )
                
                await asyncio.sleep(10.0)
                
            except Exception as e:
                logger.error(f"[RiskLoop] Error: {e}")
                await asyncio.sleep(10.0)
        
        logger.info("[Loop] Risk monitor stopped")
    
    async def _gp_evolution_loop(self):
        """
        Run genetic programming evolution in background.
        
        Every 30 minutes:
        1. Evaluate current population on recent market data
        2. Breed next generation
        3. Compile top strategies to WASM
        4. Promote best strategy to live quoting
        """
        logger.info("[Loop] GP evolution started")
        
        while not self._shutdown_event.is_set():
            try:
                # In production:
                # 1. Load recent market data
                # 2. Run GP evaluation with WASM sandbox
                # 3. Select best strategy by Sharpe
                # 4. Update gravity's fair probability model
                
                logger.info("[GP] Evolution cycle complete (placeholder)")
                await asyncio.sleep(1800.0)  # 30 minutes
                
            except Exception as e:
                logger.error(f"[GPLoop] Error: {e}")
                await asyncio.sleep(300.0)
        
        logger.info("[Loop] GP evolution stopped")
    
    async def _audit_loop(self):
        """Write audit records every 60 seconds."""
        while not self._shutdown_event.is_set():
            try:
                self._write_audit({
                    "event": "heartbeat",
                    "timestamp": time.time(),
                    "pnl": self.daily_pnl,
                    "quotes": self.quote_count,
                    "fills": self.fill_count,
                    "latency_p99": np.percentile(self.latency_samples, 99) if self.latency_samples else 0,
                })
                await asyncio.sleep(60.0)
            except Exception as e:
                logger.error(f"[AuditLoop] Error: {e}")
                await asyncio.sleep(60.0)
    
    # ------------------------------------------------------------------
    # Actions
    # ------------------------------------------------------------------
    
    async def _hedge_on_deribit(self, market_id: str, side: str, size: float) -> dict:
        """
        Cross-hedge inventory on Deribit perp.
        
        This is the callback passed to GravityMarketMaker.
        """
        if self.cfg.paper_trade:
            logger.info(f"[PAPER HEDGE] {market_id} {side} {size:.1f} on Deribit")
            return {"paper": True, "side": side, "size": size}
        
        # In production:
        # return await self.deribit_client.place_perp_order(side=side, size=size)
        logger.info(f"[HEDGE] {market_id} {side} {size:.1f} on Deribit")
        return {"executed": True}
    
    async def _emergency_stop(self):
        """Cancel all orders and halt trading."""
        logger.critical("EMERGENCY STOP ACTIVATED")
        
        for market_id in self.cfg.markets:
            if self.gravity:
                await self.gravity._cancel_all(market_id)
        
        self._write_audit({
            "event": "emergency_stop",
            "reason": "risk_limit",
            "pnl": self.daily_pnl,
            "timestamp": time.time()
        })
        
        self._shutdown_event.set()
    
    def _write_audit(self, record: dict):
        """Append to audit log with chain hashing."""
        os.makedirs(os.path.dirname(self.cfg.audit_log_path), exist_ok=True)
        
        record["_seq"] = getattr(self, '_audit_seq', 0)
        self._audit_seq = record["_seq"] + 1
        
        # Simple chain hash
        record_str = json.dumps(record, sort_keys=True)
        record["_hash"] = hashlib.sha256(record_str.encode()).hexdigest()[:16]
        
        with open(self.cfg.audit_log_path, "a") as f:
            f.write(json.dumps(record) + "\n")
    
    # ------------------------------------------------------------------
    # Diagnostics
    # ------------------------------------------------------------------
    
    def get_dashboard(self) -> dict:
        """Return a JSON-serializable status dashboard."""
        return {
            "engine": "Omega v6 (Practical)",
            "running": self.is_running,
            "paper_trade": self.cfg.paper_trade,
            "session_duration_sec": round(time.time() - self.session_start, 1),
            "pnl": {
                "daily": round(self.daily_pnl, 2),
                "peak": round(self.peak_pnl, 2),
                "drawdown": round(self.peak_pnl - self.daily_pnl, 2)
            },
            "performance": {
                "quotes": self.quote_count,
                "fills": self.fill_count,
                "latency_p99_ms": round(np.percentile(self.latency_samples, 99), 2) if self.latency_samples else 0,
            },
            "surface": {
                "events_tracked": len(self.surface.get_all_events()) if self.surface else 0,
            },
            "gravity": self.gravity.get_status() if self.gravity else {},
            "wasm": {
                "cache_size": self.wasm.cache_size if self.wasm else 0,
                "fuel_limit": self.cfg.wasm_fuel_limit
            }
        }


# ------------------------------------------------------------------------------
# CLI Entry Point
# ------------------------------------------------------------------------------

async def main():
    """Run the Omega Engine from command line."""
    config = OmegaConfig()
    
    # Override from env
    config.paper_trade = os.getenv("PAPER_TRADE", "true").lower() == "true"
    
    engine = OmegaEngine(config)
    
    try:
        await engine.start()
    except asyncio.CancelledError:
        pass
    finally:
        await engine.stop()


if __name__ == "__main__":
    asyncio.run(main())
